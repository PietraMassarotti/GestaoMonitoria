CREATE TABLE [dbo].[tb_alunos] (
    [ra]    INT           NOT NULL,
    [nome]  VARCHAR (100) NOT NULL,
    [email] VARCHAR (120) NOT NULL,
    [senha] VARCHAR (100) NOT NULL,
    PRIMARY KEY CLUSTERED ([ra] ASC),
    UNIQUE NONCLUSTERED ([email] ASC)
);

CREATE TABLE [dbo].[tb_coordenadores] (
    [id_coordenador] INT           IDENTITY (1, 1) NOT NULL,
    [nome]           VARCHAR (100) NOT NULL,
    [email]          VARCHAR (120) NOT NULL,
    [senha]          VARCHAR (100) NOT NULL,
    PRIMARY KEY CLUSTERED ([id_coordenador] ASC),
    UNIQUE NONCLUSTERED ([email] ASC)
);

CREATE TABLE [dbo].[tb_disciplinas] (
    [id_disciplina]   INT           IDENTITY (1, 1) NOT NULL,
    [nome_disciplina] VARCHAR (100) NOT NULL,
    PRIMARY KEY CLUSTERED ([id_disciplina] ASC)
);

CREATE TABLE [dbo].[tb_monitores] (
    [rm]                  INT           NOT NULL,
    [nome]                VARCHAR (100) NOT NULL,
    [email]               VARCHAR (120) NOT NULL,
    [senha]               VARCHAR (100) NOT NULL,
    [id_disciplina]       INT           NOT NULL,
    [periodo]             VARCHAR (30)  NULL,
    [data_inicio_periodo] DATE          NULL,
    [data_fim_periodo]    DATE          NULL,
    PRIMARY KEY CLUSTERED ([rm] ASC),
    UNIQUE NONCLUSTERED ([email] ASC),
    CONSTRAINT [FK_tb_monitores_tb_disciplinas] FOREIGN KEY ([id_disciplina]) REFERENCES [dbo].[tb_disciplinas] ([id_disciplina])
);

CREATE TABLE [dbo].[tb_disponibilidades] (
    [id_disponibilidade] INT          IDENTITY (1, 1) NOT NULL,
    [rm]                 INT          NOT NULL,
    [dia_semana]         VARCHAR (20) NOT NULL,
    [horario]            VARCHAR (20) NOT NULL,
    PRIMARY KEY CLUSTERED ([id_disponibilidade] ASC),
    CONSTRAINT [FK_tb_disponibilidades_tb_monitores] FOREIGN KEY ([rm]) REFERENCES [dbo].[tb_monitores] ([rm])
);

CREATE TABLE [dbo].[tb_agendamentos] (
    [id_agendamento]   INT          IDENTITY (1, 1) NOT NULL,
    [ra]               INT          NOT NULL,
    [rm]               INT          NOT NULL,
    [id_disciplina]    INT          NOT NULL,
    [dia_semana]       VARCHAR (20) NOT NULL,
    [data_agendamento] DATE         NOT NULL,
    [horario]          VARCHAR (20) NOT NULL,
    [status]           VARCHAR (20) DEFAULT ('Agendada') NOT NULL,
    [data_criacao]     DATETIME     DEFAULT (getdate()) NOT NULL,
    PRIMARY KEY CLUSTERED ([id_agendamento] ASC),
    CONSTRAINT [FK_tb_agendamentos_tb_alunos] FOREIGN KEY ([ra]) REFERENCES [dbo].[tb_alunos] ([ra]),
    CONSTRAINT [FK_tb_agendamentos_tb_monitores] FOREIGN KEY ([rm]) REFERENCES [dbo].[tb_monitores] ([rm]),
    CONSTRAINT [FK_tb_agendamentos_tb_disciplinas] FOREIGN KEY ([id_disciplina]) REFERENCES [dbo].[tb_disciplinas] ([id_disciplina])
);
